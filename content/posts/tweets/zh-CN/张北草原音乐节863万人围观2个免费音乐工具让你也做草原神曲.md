  s("pad").note("d4 f4 a4").slow(2).gain(0.5),
  note("e5 g5 a5 g5 e5 d5 e5 a5").s("lead").slow(2).gain(0.6),
  s("bd bd sd bd bd sd bd sd").gain(0.6),
  note("a2 e2 a2 e2").s("bass").slow(1)

  s("pad").note("d4").slow(4).gain(0.4),
